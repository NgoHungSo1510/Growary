## BỐI CẢNH & LÝ DO THAY ĐỔI

### V2.0 hoạt động như thế nào (hiện tại)
- Admin phải set `startTime`/`endTime` thủ công cho từng boss (221 nhân vật = rất mệt)
- Boss xuất hiện tuần tự theo lịch cố định → user đoán được → mất yếu tố bất ngờ
- `checkAndActivateBosses()` kiểm tra thời gian mỗi phút để activate boss theo lịch
- Chỉ có **1 boss active** tại một thời điểm

### V2.1 thay đổi gì
- **Không cần set lịch** — boss được bốc ngẫu nhiên mỗi tuần
- **2 boss active song song** mỗi tuần — user chọn đánh boss nào (tạo cảm giác cày để mở cả 2)
- **Pool system**: boss chưa bị hạ gục (HP > 0 khi hết tuần) → trả về pool → shuffle lại → bốc tiếp sau
- **Boss hạ gục (HP = 0)** → `completed` → không bao giờ xuất hiện lại
- **Collection chỉ để thống kê gallery** — không ảnh hưởng đến lịch xuất hiện
- **Lazy evaluation**: rotation tự kích hoạt khi user truy cập, KHÔNG phụ thuộc cron đúng giờ

### Vấn đề backend ngủ đông
Backend deploy trên hosting có thể ngủ khi không có traffic (Render free tier...).
Cron job sẽ không chạy nếu server đang ngủ.

**Giải pháp**: Gọi `checkWeeklyRotation()` ngay bên trong `GET /events/boss/active`.
Khi user đầu tiên mở app sau khi tuần mới bắt đầu → rotation tự động chạy.

> Pattern này đã được dùng thành công: `checkAndActivateBosses()` được gọi
> cả trong cron lẫn trong `eventController.ts` (line 11 và 62).

```typescript
// Trả về thứ Hai 00:00:00 của tuần hiện tại (UTC+7)
export const getStartOfWeek = (): Date => {
    const now = new Date();
    const vnOffset = 7 * 60 * 60 * 1000;
    const vnNow = new Date(now.getTime() + vnOffset);
    const day = vnNow.getUTCDay(); // 0=CN, 1=T2... 6=T7
    const diff = (day === 0 ? -6 : 1 - day);
    const monday = new Date(vnNow);
    monday.setUTCDate(vnNow.getUTCDate() + diff);
    monday.setUTCHours(0, 0, 0, 0);
    return new Date(monday.getTime() - vnOffset);
};
```

**Thêm `checkWeeklyRotation()` — HÀM QUAN TRỌNG NHẤT:**

```typescript
import { SystemConfig } from '../models/SystemConfig';

export const checkWeeklyRotation = async (): Promise<void> => {
    try {
        const currentWeekStart = getStartOfWeek();

        // Đọc lần rotate cuối từ DB
        const config = await SystemConfig.findOne({ key: 'lastBossRotation' });
        const lastRotation = config?.value ? new Date(config.value) : null;

        // Nếu đã rotate tuần này → không làm gì (idempotent)
        if (lastRotation && lastRotation >= currentWeekStart) return;

        console.log('🔄 New week detected — performing boss rotation...');

        // Atomic lock để tránh race condition (nhiều user vào cùng lúc)
        const lockResult = await SystemConfig.findOneAndUpdate(
            {
                key: 'rotationLock',
                $or: [
                    { value: 'unlocked' },
                    { updatedAt: { $lt: new Date(Date.now() - 60000) } } // lock expire 60s
                ]
            },
            { value: 'locked' },
            { upsert: true, new: true }
        ).catch(() => null);

        if (!lockResult) {
            console.log('🔒 Rotation already in progress by another request, skipping.');
            return;
        }

        await performWeeklyRotation(currentWeekStart);

        // Cập nhật thời gian rotate + release lock
        await SystemConfig.findOneAndUpdate(
            { key: 'lastBossRotation' },
            { value: currentWeekStart.toISOString() },
            { upsert: true }
        );
        await SystemConfig.findOneAndUpdate(
            { key: 'rotationLock' },
            { value: 'unlocked' }
        );

        console.log('✅ Boss rotation complete.');
    } catch (error) {
        console.error('❌ Error in checkWeeklyRotation:', error);
        // Release lock nếu lỗi
        await SystemConfig.findOneAndUpdate({ key: 'rotationLock' }, { value: 'unlocked' }).catch(() => {});
    }
};
```

**Thêm `performWeeklyRotation()`:**

```typescript
export const performWeeklyRotation = async (weekStart: Date): Promise<void> => {
    // BƯỚC 1: Kết thúc 2 boss đang active
    const activeBosses = await BossEvent.find({ status: 'active' });

    for (const boss of activeBosses) {
        if (boss.currentHp <= 0) {
            // Đã completed trong tuần, giữ nguyên
            console.log(`  ✅ "${boss.title}" was completed this week.`);
        } else {
            // Chưa bị hạ → trả về pool, reset HP
            boss.status = 'pool';
            boss.currentHp = boss.maxHp;
            boss.timesReturned = (boss.timesReturned || 0) + 1;
            boss.isRewardDistributed = false;
            boss.weekActivatedAt = undefined;
            await boss.save();
            console.log(`  🔁 "${boss.title}" returned to pool (timesReturned: ${boss.timesReturned})`);
        }
    }

    // BƯỚC 2: Bốc 2 boss mới từ pool
    // Ưu tiên boss chưa bao giờ được bốc (timesReturned = 0)
    const pool = await BossEvent.find({ status: 'pool' }).sort({ timesReturned: 1 });

    if (pool.length === 0) {
        console.log('⚠️ Boss pool is empty! No bosses to activate.');
        return;
    }

    // Shuffle để random
    const shuffled = [...pool].sort(() => Math.random() - 0.5);
    const toActivate = shuffled.slice(0, Math.min(2, shuffled.length));

    for (const boss of toActivate) {
        boss.status = 'active';
        boss.weekActivatedAt = weekStart;
        await boss.save();
        console.log(`  🎯 Activated: "${boss.title}"`);
    }
};
```

```typescript
export const getActiveBosses = async (req: AuthRequest, res: Response): Promise<void> => {
    try {
        await checkWeeklyRotation(); // Lazy rotation

        const activeBossDocs = await BossEvent.find({ status: 'active' })
            .populate('collectionId', 'themeColor title iconEmoji');

        if (activeBossDocs.length === 0) {
            res.json({ activeBosses: [], userRecords: [] });
            return;
        }

        const result = await Promise.all(activeBossDocs.map(async (bossDoc) => {
            const boss = bossDoc.toObject() as any;
            if (bossDoc.collectionId && (bossDoc.collectionId as any).themeColor) {
                boss.colorBg = (bossDoc.collectionId as any).themeColor;
                boss.collectionInfo = bossDoc.collectionId;
            }

            let userRecord = await BossRecord.findOne({ eventId: boss._id, userId: req.userId });
            if (!userRecord) {
                userRecord = await BossRecord.create({ eventId: boss._id, userId: req.userId });
            }

            return { boss, userRecord };
        }));

        res.json({
            activeBosses: result.map(r => r.boss),
            userRecords: result.map(r => r.userRecord),
        });
    } catch (error) {
        console.error('Failed to get active bosses:', error);
        res.status(500).json({ error: 'Internal server error' });
    }
};
```

```typescript
export const attackBoss = async (req: AuthRequest, res: Response): Promise<void> => {
    try {
        // THÊM bossId
        const { bossId, miniGameType, miniGameResult, usePercentage, questionIndex, answerIndex } = req.body;

        if (!bossId) {
            res.status(400).json({ error: 'bossId is required' });
            return;
        }

        const validPercentages = [0.25, 0.5, 0.75, 1.0];
        if (!validPercentages.includes(usePercentage)) {
            res.status(400).json({ error: 'Invalid usePercentage' });
            return;
        }

        // SỬA: tìm đúng boss theo bossId
        const activeBoss = await BossEvent.findOne({ _id: bossId, status: 'active' });
        if (!activeBoss) { res.status(404).json({ error: 'Boss not found or not active' }); return; }
        if (activeBoss.currentHp <= 0) { res.status(400).json({ error: 'Character already unlocked' }); return; }

        let userRecord = await BossRecord.findOne({ eventId: activeBoss._id, userId: req.userId });
        if (!userRecord || userRecord.attackPoints <= 0) {
            res.status(400).json({ error: 'No attack points available' }); return;
        }

        // --- Phần còn lại GIỮ NGUYÊN từ V2.0 ---
        let actualMiniGameResult = miniGameResult;
        if (miniGameType === 'quiz' && questionIndex !== undefined && answerIndex !== undefined) {
            const q = activeBoss.miniGameQuestions?.[questionIndex];
            actualMiniGameResult = (q && q.correctIndex === answerIndex) ? 'critical' : 'normal';
        }

        const pointsToUse = Math.floor(userRecord.attackPoints * usePercentage);
        if (pointsToUse <= 0) { res.status(400).json({ error: 'Not enough attack points' }); return; }

        const multiplier = actualMiniGameResult === 'critical' ? 1.5 : 0.75;
        const potentialDamage = Math.floor(pointsToUse * multiplier);

        let actualDamage = potentialDamage;
        let refundPoints = 0;

        if (potentialDamage > activeBoss.currentHp) {
            actualDamage = activeBoss.currentHp;
            refundPoints = Math.floor((potentialDamage - actualDamage) / multiplier);
        }

        activeBoss.currentHp -= actualDamage;
        if (activeBoss.currentHp <= 0) {
            activeBoss.currentHp = 0;
            activeBoss.status = 'completed';
        }
        await activeBoss.save();

        userRecord.attackPoints = (userRecord.attackPoints - pointsToUse) + refundPoints;
        userRecord.totalDamageDealt += actualDamage;
        userRecord.lastMiniGameType = miniGameType;
        userRecord.lastMiniGameResult = actualMiniGameResult;
        await userRecord.save();

        if (activeBoss.status === 'completed') {
            const { distributeBossRewards } = await import('../services/bossService');
            distributeBossRewards(activeBoss._id.toString()).catch(console.error);
        }

        res.json({
            success: true, actualDamage, refundPoints,
            isCritical: actualMiniGameResult === 'critical',
            bossCurrentHp: activeBoss.currentHp, bossStatus: activeBoss.status,
            userAttackPoints: userRecord.attackPoints, userTotalDamage: userRecord.totalDamageDealt,
        });
    } catch (error) {
        console.error('Failed to attack boss:', error);
        res.status(500).json({ error: 'Internal server error' });
    }
};
```

```typescript
// History timeline
router.get('/boss-weekly-history', adminMiddleware, async (req, res) => {
    const bosses = await BossEvent.find({ weekActivatedAt: { $ne: null } })
        .populate('collectionId', 'title iconEmoji')
        .sort({ weekActivatedAt: -1 });

    // Group theo tuần
    const weekMap = new Map<string, BossEvent[]>();
    for (const boss of bosses) {
        const weekKey = boss.weekActivatedAt!.toISOString().slice(0, 10);
        if (!weekMap.has(weekKey)) weekMap.set(weekKey, []);
        weekMap.get(weekKey)!.push(boss);
    }

    const weeks = Array.from(weekMap.entries()).map(([weekStart, bosses]) => ({
        weekStart,
        weekLabel: `W${getWeekNumber(new Date(weekStart))}/${new Date(weekStart).getFullYear()}`,
        bosses,
    }));

    res.json({ weeks });
});

// Pool status
router.get('/boss-pool-status', adminMiddleware, async (req, res) => {
    const [poolCount, activeCount, completedCount] = await Promise.all([
        BossEvent.countDocuments({ status: 'pool' }),
        BossEvent.countDocuments({ status: 'active' }),
        BossEvent.countDocuments({ status: 'completed' }),
    ]);
    res.json({ poolCount, activeCount, completedCount, total: poolCount + activeCount + completedCount });
```

```typescript
const handleAttackPress = async (bossId: string) => {
    const bossIdx = bosses.findIndex(b => b._id === bossId);
    const record = records[bossIdx];
    const boss = bosses[bossIdx];
    if (!record || record.attackPoints <= 0) return;

    setCurrentAttackBossId(bossId);

    const hasQuiz = boss?.miniGameQuestions?.length > 0;
    const options = hasQuiz ? ['tap', 'quiz', 'reflex'] : ['tap', 'reflex'];
    const selectedType = options[Math.floor(Math.random() * options.length)] as 'tap' | 'quiz' | 'reflex';
    setMiniGameType(selectedType);

        const qRes = await apiService.get(`/events/boss/quiz?bossId=${bossId}`);
        setQuizData(qRes);

    setIsMiniGameVisible(true);
};
```

## PHASE 7 — SEED SCRIPT

**Ước lượng: 30 phút**

### 7.1 [NEW] `backend/src/scripts/seedBossV21.ts`

```typescript
import mongoose from 'mongoose';
import dotenv from 'dotenv';
import { BossCollection } from '../models/BossCollection';
import { BossEvent } from '../models/BossEvent';
import { SystemConfig } from '../models/SystemConfig';
import { collectionsData, bossesData } from './seedFullCharacterDataBackup';

dotenv.config();
const MONGO_URI = process.env.MONGODB_URI || 'mongodb://localhost:27017/growary';

const seed = async () => {
    await mongoose.connect(MONGO_URI);
    console.log('Connected to MongoDB');

    // Xóa data cũ
    await BossEvent.deleteMany({});
    await BossCollection.deleteMany({});
    await SystemConfig.deleteOne({ key: 'lastBossRotation' });
    console.log('Cleared old boss data');

    // 1. Seed Collections
    const collectionMap: Record<string, mongoose.Types.ObjectId> = {};
    for (const col of collectionsData) {
        const { _refId, ...colData } = col as any;
        const created = await BossCollection.create(colData);
        collectionMap[_refId] = created._id;
        console.log(`  Collection: ${col.title}`);
    }

    // 2. Seed Boss Events — tất cả status = 'pool', bỏ startTime/endTime
    let count = 0;
    for (const boss of bossesData) {
        const { collectionRefId, startTime, endTime, ...bossData } = boss as any;
        await BossEvent.create({
            ...bossData,
            status: 'pool',
            currentHp: boss.maxHp,
            collectionId: collectionMap[collectionRefId] || null,
            timesReturned: 0,
            isRewardDistributed: false,
        });
        count++;
    }

    console.log(`\n✅ Seeded ${collectionsData.length} collections, ${count} characters into pool.`);
    console.log('Next: Call GET /events/boss/active to trigger first rotation (2 bosses will activate).');
    process.exit(0);
};

seed().catch(err => { console.error(err); process.exit(1); });
```
